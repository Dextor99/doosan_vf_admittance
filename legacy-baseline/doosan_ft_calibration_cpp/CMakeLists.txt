cmake_minimum_required(VERSION 3.12)
project(doosan_ft_calibration LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(ft_calibration src/ft_calibration.cpp)
target_include_directories(ft_calibration PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
    target_compile_options(ft_calibration PRIVATE /W4)
else()
    target_compile_options(ft_calibration PRIVATE -Wall -Wextra -Wpedantic -O2)
endif()

# Offline tools remain buildable without the vendor SDK.
option(BUILD_DOOSAN_ACQUISITION "Build Linux Doosan/XJC read-only acquisition" ON)
set(DOOSAN_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../BINARY_GL010118_20230324A" CACHE PATH "Doosan SDK directory")
set(DOOSAN_SDK_LIB "${DOOSAN_SDK_ROOT}/library/Linux/64bits/20.04")
if(BUILD_DOOSAN_ACQUISITION AND CMAKE_SYSTEM_NAME STREQUAL "Linux" AND
   EXISTS "${DOOSAN_SDK_ROOT}/include/DRFLEx.h" AND EXISTS "${DOOSAN_SDK_LIB}/libDRFL.a")
    add_executable(ft_acquire src/ft_acquire.cpp)
    target_include_directories(ft_acquire PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    target_include_directories(ft_acquire SYSTEM PRIVATE ${DOOSAN_SDK_ROOT}/include)
    target_compile_options(ft_acquire PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(ft_acquire PRIVATE "${DOOSAN_SDK_LIB}/libDRFL.a"
        "${DOOSAN_SDK_LIB}/libPocoNet.so.62" "${DOOSAN_SDK_LIB}/libPocoFoundation.so.62" pthread)
    set_target_properties(ft_acquire PROPERTIES BUILD_RPATH "${DOOSAN_SDK_LIB}")
    set_property(TARGET ft_acquire APPEND_STRING PROPERTY LINK_FLAGS " -Wl,--disable-new-dtags")
    add_executable(ft_auto src/ft_auto.cpp)
    target_include_directories(ft_auto PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    target_include_directories(ft_auto SYSTEM PRIVATE ${DOOSAN_SDK_ROOT}/include)
    target_compile_options(ft_auto PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(ft_auto PRIVATE "${DOOSAN_SDK_LIB}/libDRFL.a"
        "${DOOSAN_SDK_LIB}/libPocoNet.so.62" "${DOOSAN_SDK_LIB}/libPocoFoundation.so.62" pthread)
    set_target_properties(ft_auto PROPERTIES BUILD_RPATH "${DOOSAN_SDK_LIB}")
    set_property(TARGET ft_auto APPEND_STRING PROPERTY LINK_FLAGS " -Wl,--disable-new-dtags")
    add_executable(ft_record_poses src/ft_record_poses.cpp)
    target_include_directories(ft_record_poses SYSTEM PRIVATE ${DOOSAN_SDK_ROOT}/include)
    target_compile_options(ft_record_poses PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(ft_record_poses PRIVATE "${DOOSAN_SDK_LIB}/libDRFL.a"
        "${DOOSAN_SDK_LIB}/libPocoNet.so.62" "${DOOSAN_SDK_LIB}/libPocoFoundation.so.62" pthread)

    add_executable(doosan_movej_autoinit src/doosan_movej_autoinit.cpp)
    target_include_directories(doosan_movej_autoinit SYSTEM PRIVATE ${DOOSAN_SDK_ROOT}/include)
    target_compile_options(doosan_movej_autoinit PRIVATE -Wall -Wextra -Wpedantic)
